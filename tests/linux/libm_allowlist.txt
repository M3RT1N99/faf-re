# Engine addition (dedicated-server plan WPL.13): the faf_sim functions that may call a host libm
# transcendental (tests/linux/LibmGate.py). One "symbol demangled-function" per line; the reason
# above each group. A new call fails the Linux build until it is reviewed here.

# legacy/math/X87Math: the CRT-dispatch lanes. In the domain the engine's x87 bodies run
# (X87Math64.S); the host CRT is reached only with the CRT's SSE2 mode on (never in the sim) or
# for pow's zero, infinite and NaN operands (IEEE-defined results). asinf, expf and logf use the
# engine's legacy/math/Libm.cpp (plan WPL.11) and are not listed: a host call there fails.
acos msvc8::acos(double)
# The same acos lane, inlined into acosf by the Android NDK's clang 21.
acos msvc8::acosf(float)
asin msvc8::asin(double)
pow msvc8::pow(double, double)
# powf: the sim driver's wall-clock pacing only (SimDriver.cpp, the beat rate from the sim rate).
pow msvc8::powf(float, float)

# legacy/math/X87MathSoft (arm64, which has no x87 unit): the x87's transcendental instructions,
# from the C library's binary128 functions rounded to the register's 64 bits. Those are accurate
# to about 2^-112, so every C library with a binary128 long double (bionic, glibc on AArch64)
# gives the same register value; tests/sim_math checks the routines against the x87.
sinl (anonymous namespace)::SineOfQuadrant(long double, unsigned int)
cosl (anonymous namespace)::SineOfQuadrant(long double, unsigned int)
tanl (anonymous namespace)::TangentOfQuadrant(long double, unsigned int)
atan2l (anonymous namespace)::Fpatan(long double, long double)
log2l (anonymous namespace)::Fyl2x(long double, long double)
expm1l (anonymous namespace)::F2xm1(long double)

# Camera projection (effect culling against the sync filter's cameras, the render-side view):
# visual only.
tanf moho::GeomCamera3::ViewInitPerspective(float, float, float, float)
tanf moho::VEC_D3DProjectionMatrixFOV(float, float, float, float, float)

# Effects and decals: visual state, not in the sim checksum.
atan2f moho::CEfxEmitter::Tick(int)
atan2f (anonymous namespace)::CreateDecalFromTransform(moho::VTransform const&, Wm3::Vector3<float> const&, moho::Sim*, float, msvc8::string const&, msvc8::string const&, float, bool, msvc8::string const&, unsigned int, unsigned int)

# remote/wire: the server-authoritative wire format (the server quantises, clients decode); not

atan2 faf::remote::wire::QuantizeYaw(std::__1::array<float, 4ul> const&)
