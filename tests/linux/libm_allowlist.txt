# legacy/math/X87Math: the CRT-dispatch lanes. In the domain the engine's x87 bodies run
# (X87Math64.S); the host CRT is reached only with the CRT's SSE2 mode on (never in the sim) or
# for pow's zero, infinite and NaN operands (IEEE-defined results). asinf, expf and logf use the
# engine's legacy/math/Libm.cpp (plan WPL.11) and are not listed: a host call there fails.

# The same acos lane, inlined into acosf by the Android NDK's clang 21.
acos fa::acosf(float)
asin fa::asin(double)
pow fa::pow(double, double)
# powf: the sim driver's wall-clock pacing only (SimDriver.cpp, the beat rate from the sim rate).

# powf: the skew clock's tick length and display factor (SimDriver.cpp), wall-clock pacing and a log
# line; nothing it computes reaches the sim.
powf (anonymous namespace)::ApplySharedStart((anonymous namespace)::SkewClock&, int, int, moho::CClientManagerImpl&)
powf moho::CSimDriver::GetSyncData(moho::SSyncData*&)

# legacy/math/X87MathSoft (arm64, which has no x87 unit): the x87's transcendental instructions,
# from the C library's binary128 functions rounded to the register's 64 bits. Two C libraries give
# the same register value unless their results differ within an ulp of 113 bits next to a 64-bit
# rounding boundary: at most about 2^-48 per call, and a register value one 64-bit ulp off changes
# a routine's float result only next to a float rounding boundary (about 2^-40 of those). On a
# device the system's libm is used (Android links bionic dynamically), not the NDK's libm.a that a
# static test build links. tests/sim_math checks the routines against the x87.
sinl (anonymous namespace)::SineOfQuadrant(long double, unsigned int)
cosl (anonymous namespace)::SineOfQuadrant(long double, unsigned int)
tanl (anonymous namespace)::TangentOfQuadrant(long double, unsigned int)
atan2l (anonymous namespace)::Fpatan(long double, long double)
log2l (anonymous namespace)::Fyl2x(long double, long double)
expm1l (anonymous namespace)::F2xm1(long double)
